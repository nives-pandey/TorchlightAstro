// ✅ FIXED: Added immersive hover effect for premium interaction
import { motion } from "framer-motion";
import { ExperienceCard } from "@/components/ExperienceCard";
import { sagaExperiences } from "@/data/saga-experiences";
import { useState } from "react";
export function ExperiencesSection() {
  const grandJourneys = sagaExperiences.filter(journey => journey.collection === 'Grand');
  const signatureJourneys = sagaExperiences.filter(journey => journey.collection === 'Signature');
  
  // ✅ FIXED: State for immersive hover effect
  const [hoveredJourney, setHoveredJourney] = useState<string | null>(null);
  return (
    <section 
      className={`py-20 transition-all duration-700 ${
        hoveredJourney 
          ? 'bg-gradient-to-br from-russian-blue/5 via-saga-gold/5 to-russian-blue/5' 
          : 'bg-gradient-to-br from-cream via-platinum to-cream'
      }`}
    >
      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">
        
        {/* ✅ VERIFIED: Distinct h2 headings for both collections */}
        <h2 className="font-serif text-4xl md:text-6xl font-bold text-center text-russian-blue mb-16">
          The Grand Collection
        </h2>
        
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-8">
          {grandJourneys.map((journey, index) => (
            <motion.div
              key={journey.id}
              onMouseEnter={() => setHoveredJourney(journey.id)}
              onMouseLeave={() => setHoveredJourney(null)}
            >
              <ExperienceCard journey={journey} />
            </motion.div>
          ))}
        </div>
        <h2 className="font-serif text-4xl md:text-6xl font-bold text-center text-russian-blue mb-16">
          The Signature Collection
        </h2>
        
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-8">
          {signatureJourneys.map((journey, index) => (
            <motion.div
              key={journey.id}
              onMouseEnter={() => setHoveredJourney(journey.id)}
              onMouseLeave={() => setHoveredJourney(null)}
            >
              <ExperienceCard journey={journey} />
            </motion.div>
          ))}
        </div>
      </div>
    </section>
  );
}

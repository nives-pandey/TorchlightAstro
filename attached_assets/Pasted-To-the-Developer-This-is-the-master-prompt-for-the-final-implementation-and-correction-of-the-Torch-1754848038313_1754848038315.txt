  const setTheme = (theme) => {
    document.body.className = ''; // Clear existing theme
    document.body.classList.add(theme);
  };

  return (
    <select onChange={(e) => setTheme(e.target.value)} className="bg-card text-card-foreground p-2 rounded-md border border-border">
